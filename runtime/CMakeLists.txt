# The GP32 runtime every recompiled program links against.
cmake_minimum_required(VERSION 3.15)
if(NOT CMAKE_PROJECT_NAME)
  project(gp32rt C)
endif()

set(GP32RT_SOURCES cpu.c hw.c bios.c smc.c main.c)
if(WIN32 AND NOT GP32_HEADLESS)
  list(APPEND GP32RT_SOURCES platform_win32.c)
else()
  list(APPEND GP32RT_SOURCES platform_null.c)
endif()

add_library(gp32rt STATIC ${GP32RT_SOURCES})
target_include_directories(gp32rt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_property(TARGET gp32rt PROPERTY C_STANDARD 11)
if(WIN32 AND NOT GP32_HEADLESS)
  target_link_libraries(gp32rt PUBLIC user32 gdi32 winmm)
endif()
if(MSVC)
  target_compile_options(gp32rt PUBLIC /W3 /wd4100 /wd4102 /wd4702 /D_CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(gp32rt PUBLIC -Wall -Wno-unused-label -fno-strict-aliasing)
endif()
